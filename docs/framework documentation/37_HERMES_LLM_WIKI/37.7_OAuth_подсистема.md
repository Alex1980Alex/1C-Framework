    """Adds legacy `_task` suffix aliases."""

    async def start_cleanup_task(self, interval: int = 60) -> None:
        await self.start_cleanup(interval)

    async def stop_cleanup_task(self) -> None:
        await self.stop_cleanup()


class OAuth2Service(_GenericOAuth2Service):
    """BSL adapter: (login, password) packed into generic `user_data`."""

    async def generate_authorization_code(
        self, login, password, redirect_uri, code_challenge,
    ) -> str:
        return await super().generate_authorization_code(
            client_id="bsl-legacy",
            redirect_uri=redirect_uri,
            code_challenge=code_challenge,
            user_data={"login": login, "password": password},
        )

    async def validate_access_token(self, token) -> tuple[str, str] | None:
        data = await super().validate_access_token(token)
        if not data:
            return None
        return (data["login"], data["password"])

### Routes ([`routes.py`](../../../src/shared/mcp_oauth/fastapi/routes.py))

```python
def build_oauth_router() -> APIRouter:
    router = APIRouter(tags=["oauth"])

    @router.get("/.well-known/oauth-protected-resource")
    async def prm_metadata(request: Request):
        """RFC 9728 PRM."""

    @router.get("/.well-known/oauth-authorization-server")
    async def authz_server_metadata(request: Request):
        """RFC 8414 ASM."""

    @router.get("/authorize")
    async def authorize(
        request: Request,
        client_id: str = Query(...),
        redirect_uri: str = Query(...),
        code_challenge: str = Query(...),
        code_challenge_method: str = Query("S256"),
        state: str | None = Query(None),
    ):
        """PKCE S256 mandatory; rejects plain."""

    @router.post("/token")
    async def token(
        request: Request,
        grant_type: str = Form(...),
        code: str | None = Form(None),
        redirect_uri: str | None = Form(None),
        code_verifier: str | None = Form(None),
        refresh_token: str | None = Form(None),
    ):
        """Supports authorization_code + refresh_token grants."""

    return router


def _public_url(request: Request) -> str:
    """Derive public-facing URL from request, honoring proxy headers.

    SECURITY: `x-forwarded-proto` / `x-forwarded-host` trusted unconditionally.
    Deploy behind reverse proxy that strips these from inbound (nginx
    `proxy_set_header`, Envoy `request_headers_to_remove`). Otherwise client
    can spoof `issuer` in PRM/ASM metadata. Pair with TrustedHostMiddleware.
    """
```

### Dependency ([`dependencies.py`](../../../src/shared/mcp_oauth/fastapi/dependencies.py))

```python
async def require_oauth(request: Request) -> dict:
    """Bearer token validation; supports feature-flag bypass."""
    oauth_enabled = getattr(request.app.state, "oauth_enabled", True)
    if not oauth_enabled:
        return {}  # ← bypass (можно использовать как audit signal)

    auth = request.headers.get("Authorization", "")
    if not auth.startswith("Bearer "):
        raise HTTPException(401, "Missing Bearer token",
                          headers={"WWW-Authenticate": 'Bearer error="invalid_token"'})

    service = request.app.state.oauth_service
    user_data = await service.validate_access_token(auth[7:])
    if not user_data:
        raise HTTPException(401, "Invalid or expired token")
    return user_data
```

### AuditedOAuth2Service

```python
class AuditedOAuth2Service(OAuth2Service):
    """Sync sink contract — emit failures swallowed, audit never blocks auth."""

    def __init__(self, *args, audit_emit: Callable[[OAuthAuditEvent], None] | None = None, **kw):
        super().__init__(*args, **kw)
        self._emit = audit_emit or _noop

    def _safe_emit(self, event: OAuthAuditEvent) -> None:
        try:
            self._emit(event)
        except Exception:  # noqa: BLE001 — audit never blocks auth
            pass

    async def generate_authorization_code(self, ...) -> str:
        code = await super().generate_authorization_code(...)
        self._safe_emit(OAuthAuditEvent(event_type="code_issued", ...))
        return code

    # Similar overrides for exchange / refresh / validate
```

### Security non-leaks (verified by reviewer)

- Events carry only `client_id`, `event_type`, `success`, generic `failure_reason`
- **NO** `access_token`, `refresh_token`, `code`, `code_verifier`, `code_challenge` values in payloads
- `failure_reason` strings are generic (`invalid_or_expired_token`, `invalid_code_or_pkce_or_redirect`) — не reveal which specific check failed → mitigates oracle attacks

## Архитектурное уточнение для pdf-vector-graph

[`src/mcp_server/server.py`](../../../src/mcp_server/server.py) — **stdio-only transport**. OAuth applies только к HTTP/SSE MCP, не к stdio (local IPC). Поэтому F5.2 (планировавшаяся integration в pdf-vector-graph) переориентирована на **reusable FastAPI router**.

Когда пользователь захочет добавить HTTP transport для pdf-vector-graph — одна строка:

```python
app.include_router(build_oauth_router())
```

И настройка `app.state.oauth_service` + `app.state.oauth_enabled`. Готово.

## Antipatterns

| Плохо | Почему | Как правильно |
|---|---|---|
| Использовать BSL wrapper для нового кода | Sync-shaped legacy API, login/password coupling | Импортировать напрямую из `src.shared.mcp_oauth` с `user_data: dict` |
| Trust `x-forwarded-*` без proxy | Client может spoof `issuer` в PRM/ASM | Deploy за reverse proxy который strips inbound headers + adds outbound |
| Hardcode `MCP_OAUTH_ENABLED=true` в `.mcp.json` | Все environments forced enabled, нет feature-flag rollout | Keep `false` default, override via shell env in specific deployments |
| Subclass `OAuth2Service` для custom user_data | Лучше пакуй в `user_data: dict` (generic) | Subclass только если меняется flow (например AuditedOAuth2Service) |
| Игнорировать refresh rotation | Refresh tokens single-use, новый refresh приходит вместе с access | Всегда сохраняй новый refresh после каждого `refresh_tokens()` |
